\begin{subproblem}{}
    Provide a formula for computing the angle between two nonzero $d$-dimensional vectors $\bar{x}^{(1)}$ and $\bar{x}^{(2)}$. 
    Now suppose $\bar{x}^{(1)}= [\frac{-\sqrt{3}}{4},\frac{1}{4}]^T$ and $\bar{x}^{(1)}= [\frac{\sqrt{3}}{2},\frac{1}{2}]^T$. What is the magnitude of each vector, and what is the angle between them in degrees?
\end{subproblem}
\begin{solution}
    \begin{align*}
        \bar{x}^{(1)} \cdot \bar{x}^{(2)} 
        &= \left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\| \cos{\theta}\\
        \cos{\theta} &= \frac{\bar{x}^{(1)} \cdot \bar{x}^{(2)} }{\left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\|}\\
        \theta &= \arccos \left( \frac{\bar{x}^{(1)} \cdot \bar{x}^{(2)} }{\left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\|} \right)
    \end{align*}
    \[\left\|\bar{x}^{(1)}\right\|=\sqrt{\left( \frac{-\sqrt{3}}{4} \right)^2 + \frac{1}{4^2}} = \sqrt{\frac{1}{4}}=\frac{1}{2}\]
    \[\left\|\bar{x}^{(2)}\right\|=\sqrt{\left( \frac{\sqrt{3}}{2} \right)^2 + \frac{1}{2^2}=1}\]
    \[\theta=\arccos\left( \frac{\frac{-1}{4}}{\frac{1}{2}} \right)
    =\arccos\left(  \frac{-1}{2}\right)=120^\circ\]
\end{solution}

\begin{subproblem}{}
    Fix a $d$-dimensional vector $\barv{\theta}$ and $\bar{\theta} \neq 0$. 
    Let $V$ be the hyperplane in $\R_d$ that includes all points $\barv{x}$
    s.t. $\bar{x} \cdot \bar{\theta} = 0$. Find all of the $d$-dimensional vectors that are orthogonal to 
    $V$ and have a length of $C$. Justify your answer.
\end{subproblem}
\begin{solution}
    \setlength{\parindent}{0pt}
    Let $S_C$ be the set of all $d$-dimensional vectors that are orthogonal to 
    $V$ and have a length of $C$.

    Consider $S=\left\{ \pm \frac{C}{\left\|\bar{\theta}\right\|} \bar{\theta}\right\}$

    Let $\bar{w} \in S$ and $v \in V$.
    \[\bar{w} \cdot \bar{v} = \pm \frac{C}{\left\|\bar{\theta}\right\|}\bar{\theta} \cdot \bar{v} = 0\]
    and
    \[\left\|\bar{w}\right\| = \sqrt{\sum_{i=1}^{d} \left( \pm \frac{C}{\left\|\bar{\theta}\right\|}\theta_i \right)^2}=\frac{C}{\left\|\bar{\theta}\right\|}\left\|\theta\right\|=C\]
    Thus, $\bar{w} \in S_C$, so $S \subseteq S_C$

    Let $\bar{w} \in S_C$, $\lambda = \frac{\bar{w} \cdot \bar{\theta}}{\left\|\bar{\theta}\right\|^2}$, and $\bar{u}=\bar{w}-\lambda\bar{\theta}$.
    \[\bar{u} \cdot \bar{\theta}=\left( \bar{w}-\lambda\bar{\theta} \right) \cdot \bar{\theta}=\bar{w} \cdot \bar{\theta} - \lambda\left( \bar{\theta} \cdot \bar{\theta} \right)= \bar{w} \cdot \bar{\theta} - \bar{w} \cdot \bar{\theta} = 0 \]
    Thus, $\bar{u} \in V$, and since $\bar{w} \in S_C$ this implies $\bar{u} \cdot \bar{w} = 0$.
    \[0=\bar{u} \cdot \bar{w}=\left( \lambda\theta + \bar{u} \right) \cdot \bar{u} = \lambda\left( \bar{\theta} \cdot \bar{u} \right) + \left( \bar{u} \cdot \bar{u} \right) = \left( \bar{u} \cdot \bar{u} \right) = \left\| \bar{u}\right\|^2 \]
    \[\left\| \bar{u}\right\| = 0 \implies \bar{u} = 0\]
    Thus, $\bar{w} = \lambda \bar{\theta}$, and since $\left\|\bar{w}\right\|=C$
    \begin{align*}
        C=\left\|\bar{w}\right\| &= \left\|\lambda \bar{\theta}\right\|\\ 
        &= \left| \lambda \right| \left\|\bar{\theta}\right\|\\
        \left| \lambda \right| &= \frac{C}{\left\|\bar{\theta}\right\|}
        \lambda = \pm \frac{C}{\left\|\bar{\theta}\right\|}
    \end{align*}
    Thus, $\bar{w} \in S$, so $S_C \subseteq S$.

    Therefore, $S=S_C$.
\end{solution}

\begin{subproblem}{}
    Consider an arbitrary $d$-dimensional point $\bar{x} \in \R^d$ and
a hyperplane through the origin parameterized by $\barv{\theta}$. The signed distance of $\bar{x}$ from the
plane is the perpendicular distance between $\bar{x}$ and the hyperplane (distance between $\bar{x}$ and its projection
on the hyperplane), multiplied by +1 if $\bar{x}$ lies on the same side of the plane as the vector $\bar{\theta}$ points and
by $+1$ if $\bar{x}$ lies on the opposite side.
\end{subproblem}
\begin{subsubproblem}{}
    Derive the expression for the signed distance of $\bar{x}$ from the hyperplane.
\end{subsubproblem}
\begin{solution}{}
    \setlength{\parindent}{0pt}
    From linear algebra we can write $\bar{x} \in \R_d$ uniquely in orthogonal components,
    \[\bar{x}=\bar{x}_{||}+\bar{x}_{\bot}\]
    where $\bar{x}_{||}$ is parallel to $\bar{\theta}$ and $\bar{x}_\bot$ is orthogonal to $\bar{\theta}$.

    Since $\bar{x}_{||}$ is parallel to $\bar{\theta}$
    \[\bar{x}_{||}=\lambda\bar{\theta}\]
    s.t. $\lambda \in \R_d$.
    \begin{align*}
        \bar{\theta} \cdot \bar{x} = \bar{\theta} \cdot (\bar{x}_{||}+\bar{x}_{\bot}) = \bar{\theta} \cdot \bar{x}_{||} + \bar{\theta} \cdot \bar{x}_{\bot} &= \bar{\theta} \cdot \bar{x}_{||}\\
        &=\lambda \bar{\theta} \cdot \bar{\theta}\\
        &=\lambda \left\| \bar{\theta}\right\|^2\\
        \lambda &= \frac{\bar{\theta}\cdot \bar{x}}{\left\| \bar{\theta}\right\|^2}
    \end{align*}

    Therefore, $\left\|\bar{x}_{||}\right\|=\frac{\left| \bar{x} \cdot \bar{\theta} \right|}{\left\|\bar{\theta}\right\|}$ is the distance of $\bar{x}$ from the hyperplane.

    But we wanted the signed distance, but since of $\bar{x} \cdot \bar{\theta} > 0$ when $\bar{x}$ and $\bar{\theta}$ are on the same size and negative when on opposite sides, we can just remove the absolute value bar.
    Thus, the signed distance is
    \[\frac{\bar{\theta} \cdot \bar{x}}{\left\|\theta\right\|}\]
\end{solution}

\begin{subsubproblem}{}
    Now consider a hyperplane with offset (i.e. $\bar{\theta} \cdot \bar{x} + b = 0$). How does the solution from part
(i) change?
\end{subsubproblem}
\begin{solution}{}
    \setlength{\parindent}{0pt}
    Let $\bar{x}_0$ be on the hyperplane. Then we can do the same thing as in (i) but treat $\bar{x}_0$ as the origin, so we look at the vector $\bar{x}-\bar{x}_0$.

    Thus, from $(i)$ we get the signed distance is
    \[\frac{\bar{\theta} \cdot \bar{x}-\bar{x}_0}{\left\|\theta\right\|}=\frac{\bar{\theta} \cdot \bar{x} - \bar{\theta} \cdot \bar{x}_0}{\left\|\theta\right\|}=\frac{\bar{\theta} \cdot \bar{x} + b}{\left\|\theta\right\|}\]
\end{solution}

\begin{problem}{Perceptron Algorithm with Offset}

    \begin{minted}{python}
        def perceptron(X: np.array, y: np.array):
        """
        Implements the Perception algorithm for binary linear classification.
        Args:
            X: np.array, shape (n, d) 
            y: np.array, shape (n,)

        Returns:
            theta: np.array, shape (d,)
            b: float
            alpha: np.array, shape (n,). 
                Misclassification vector, in which the i-th element is has the number of times 
                the i-th point has been misclassified)
        """
        n, d = X.shape
        theta = np.zeros((d,))
        b = 0
        alpha = np.zeros((n,))

        while not all_correct(X,y,theta,b):
            for i,(Xi,yi) in enumerate(zip(X,y)):
                if yi * (theta @ Xi + b) <= 0:
                    theta += yi * Xi
                    b += yi
                    alpha[i]+=1
        return theta, b, alpha
    \end{minted}
\end{solution}

\begin{subproblem}{}
    Given a set of linearly separable points, does the order in which the points are presented to the
algorithm affect whether or not the algorithm will converge?
\end{subproblem}
\begin{solution}{}
    No, convergence depends on linear separability of the data, which is independent of the order points are presented.
\end{solution}

\begin{subproblem}{}
    In general, assuming the algorithm converges, could changing the order in which data points are
considered change the solution to which the algorithm converges
\end{subproblem}
\begin{solution}
    Yes, different orderings produce different update sequences and can converge to different separating hyperplanes.
\end{solution}

\begin{subproblem}{}
    Given a separating hyperplane parameterized by $\bar{\theta}^{*'} \in \R^d, \|\bar{\theta}^{*'}\| \neq 0$, 
    prove that we can parameterize this same hyperplane by $\bar{\theta}^*$ s.t. $\|\bar{\theta}^*\| = 1$.
\end{subproblem}
\begin{problemproof}
    Let \[\bar{\theta}^{*} = \frac{\bar{\theta}^{*'}}{\|\bar{\theta}^{*'}\|}\]
    \begin{align*}
        \|\bar{\theta}^{*}\| = \frac{\|\bar{\theta}^{*'}\|}{\|\bar{\theta}^{*'}\|} = 1
    \end{align*}
    
    Let $\bar{x}$ be on the hyperplane described by $\bar{\theta}^{*'}$. Consider,
    \begin{align*}
        \bar{\theta}^{*} \cdot \bar{x} = \|\bar{\theta}^{*'}\|( \bar{\theta}^{*'} \cdot \bar{x} ) = 0
    \end{align*}
    and by similar argument there will be no extra vectors that will be zero since if 
    $\bar{\theta}^{*'} \cdot \bar{x} \neq 0$ then $\bar{\theta}^{*} \cdot \bar{x} \neq 0$.

    Thus, $\bar{\theta}^{*}$ parameterizes the same hyperplane.
\end{problemproof}

\begin{subproblem}{}
    Now, prove that we can rescale all of our data points so that $\|\bar{x}^{(i)}\| \leq 1, \forall i = 1, \dots, n$ without
    changing the classification of any of the points with respect to the hyperplane defined by $\bar{\theta}^{*}$.
\end{subproblem}
\begin{problemproof}
    Let \[\bar{x}^{(i)*} = \frac{\bar{x}^{(i)}}{\|\bar{x}^{(i)}\|}\]
    for all $i \in \{1,\dots,n\}$
    \[\|\bar{x}^{(i)*}\| = 1 \leq 1\]
    If $\bar{y}^{(i)}\bar{\theta}^{*'} \cdot \bar{x}^{(i)} > 0$ then
    \[\bar{y}^{(i)}\bar{\theta}^{*} \cdot \bar{x}^{(i)*} = \|\bar{\theta}^{*'}\| \cdot \|\bar{x}^{(i)}\|( \bar{\theta}^{*'} \cdot \bar{x}^{(i)} ) > 0\]
    and if $\bar{y}^{(i)}\bar{\theta}^{*'} \cdot \bar{x}^{(i)} < 0$ then
    \[\bar{y}^{(i)}\bar{\theta}^{*} \cdot \bar{x}^{(i)*} < 0\] by the same argument.

    Therefore, since the dot products are the same sign as the originals this implies that the classification is the same.
\end{problemproof}

\begin{subsubproblem}{}
    Prove that $\bar{\theta}^{(k+1)} \cdot \bar{\theta}^* \geq \bar{\theta}^{(k)} \cdot \bar{\theta}^* + \gamma$.
\end{subsubproblem}
\begin{problemproof}
    Let $\bar{x}^{(i)}$ be the vector that updated $\theta^{(k)}$ to $\theta^{(k+1)}$. Then
    \[\theta^{(k+1)} = \theta^{(k)} + y^{(i)}\bar{x}^{(i)}\]
    so
    \begin{align*}
        \bar{\theta}^{(k+1)} \cdot \bar{\theta}^*
        &=(\theta^{(k)} + y^{(i)}\bar{x}^{(i)}) \cdot \bar{\theta}^*\\
        &=\theta^{(k)} \cdot \bar{\theta}^* + y^{(i)}(\bar{x}^{(i)}\cdot \bar{\theta}^*)\\
        &\geq \theta^{(k)} \cdot \bar{\theta}^* + \gamma 
        && \t{This is because } \gamma \t{ is the min of all the } y^{(i)}(\bar{x}^{(i)}\cdot \bar{\theta}^*).
    \end{align*}
\end{problemproof}

\begin{subsubproblem}{}
    Prove that  $\|\bar{\theta}^{(k+1)}\| \geq (k+1)\gamma$.
\end{subsubproblem}
\begin{problemproof}
    From $i.$ we have that $\bar{\theta}^{(k+1)} \cdot \bar{\theta}^* \geq \bar{\theta}^{(k)} \cdot \bar{\theta}^* + \gamma$. 
    Applying this formula recursively we get 
    \[\bar{\theta}^{(k+1)} \cdot \bar{\theta}^* \geq (k+1)\gamma\] since $\theta^{(0)} \cdot \bar{\theta}^* = 0$ for the last step. Thus,
    \begin{align*}
        \bar{\theta}^{(k+1)} \cdot \bar{\theta}^* 
        &\geq (k+1)\gamma\\
        \|\bar{\theta}^{(k+1)}\| \cdot \|\bar{\theta}^*\| \cos(\alpha)
        &\geq (k+1)\gamma && \t{Where } \alpha \t{ is the angle between } \bar{\theta}^{(k+1)} \t{ and } \bar{\theta}^*.\\
        \intertext{Since $|\cos(\alpha)| \leq 1$ and $\|\bar{\theta}^*\| = 1$ this implies}
        \|\bar{\theta}^{(k+1)}\| \geq \|\bar{\theta}^{(k+1)}\| \cdot \|\bar{\theta}^*\| \cos(\alpha)
        &\geq (k+1)\gamma\\
        \|\bar{\theta}^{(k+1)}\| &\geq (k+1)\gamma
    \end{align*}
\end{problemproof}

\begin{subproblem}{}
    Now we will prove an upper bound on the magnitude of $\bar{\theta}^{(k)}$.
\end{subproblem}
\setcounter{subsubproblemcounter}{0}
\begin{subsubproblem}{}
    Prove that $\|\bar{\theta}^{(k+1)}\|^2 \leq \|\bar{\theta}^{(k)} \| + 1$.
\end{subsubproblem}
\begin{problemproof}
    Let $\bar{x}^{(i)}$ be the vector that updated $\theta^{(k)}$ to $\theta^{(k+1)}$. Then
    \begin{align*}
        \|\bar{\theta}^{(k+1)}\|^2 
        &= \bar{\theta}^{(k+1)} \cdot \bar{\theta}^{(k+1)}\\
        &= (\bar{\theta}^{(k)} + y^{(i)}\bar{x}^{(i)}) \cdot (\bar{\theta}^{(k)} + y^{(i)}\bar{x}^{(i)})\\
        &= \|\bar{\theta}^{(k)}\|^2 + \|\bar{x}^{(i)}\|^2 + 2y^{(i)}(\bar{\theta}^{(k)} \cdot \bar{x}^{(i)})\\
        \intertext{Since $\|\bar{x}^{(i)}\| \leq 1$}
        &\leq \|\bar{\theta}^{(k)}\|^2 + 1 + 2y^{(i)}(\bar{\theta}^{(k)} \cdot \bar{x}^{(i)})\\
        &\leq \|\bar{\theta}^{(k)}\|^2 + 1
    \end{align*}
    Since $y^{(i)}(\bar{\theta}^{(k)} \cdot \bar{x}^{(i)}) \leq 0$\\
    Therefore, $\|\bar{\theta}^{(k+1)}\|^2 \leq \|\bar{\theta}^{(k)} \| + 1$.
\end{problemproof}
\begin{problemproof}
    Again applying the formula recursively we get $\|\bar{\theta}^{(k+1)}\|^2 \leq (k+1)$\\ 
    since $\|\theta^{(0)}\|=0$ for the last step.
\end{problemproof}

\begin{subproblem}{}
    Relate inequalities in $(c)(ii)$ and $(d)(ii)$ to prove an upper bound on k + 1 solely in terms of $\gamma$.
\end{subproblem}
\begin{problemproof}
    From $(c)(ii)$ we get
    \begin{align*}
        (k+1)\gamma &\leq \|\bar{\theta}^{(k+1)}\|\\
        \intertext{From $(d)(ii)$ we get}
        (k+1)^2 \gamma^2 &\leq \|\bar{\theta}^{(k+1)}\|^2 \leq k+1\\
        (k+1)^2 \gamma^2 &\leq k+1\\
        k + 1 &\leq \frac{1}{\gamma^2}
    \end{align*}
\end{problemproof}

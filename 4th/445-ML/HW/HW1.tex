\qh[Vectors \& Planes Review]
\begin{subproblem}{}
    Provide a formula for computing the angle between two nonzero $d$-dimensional vectors $\bar{x}^{(1)}$ and $\bar{x}^{(2)}$. 
    Now suppose $\bar{x}^{(1)}= [\frac{-\sqrt{3}}{4},\frac{1}{4}]^T$ and $\bar{x}^{(1)}= [\frac{\sqrt{3}}{2},\frac{1}{2}]^T$. What is the magnitude of each vector, and what is the angle between them in degrees?
\end{subproblem}
\begin{solution}
    \begin{align*}
        \bar{x}^{(1)} \cdot \bar{x}^{(2)} 
        &= \left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\| \cos{\theta}\\
        \cos{\theta} &= \frac{\bar{x}^{(1)} \cdot \bar{x}^{(2)} }{\left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\|}\\
        \theta &= \arccos \left( \frac{\bar{x}^{(1)} \cdot \bar{x}^{(2)} }{\left\| \bar{x}^{(1)} \right\|\left\|\bar{x}^{(2)}\right\|} \right)
    \end{align*}
    \[\left\|\bar{x}^{(1)}\right\|=\sqrt{\left( \frac{-\sqrt{3}}{4} \right)^2 + \frac{1}{4^2}} = \sqrt{\frac{1}{4}}=\frac{1}{2}\]
    \[\left\|\bar{x}^{(2)}\right\|=\sqrt{\left( \frac{\sqrt{3}}{2} \right)^2 + \frac{1}{2^2}=1}\]
    \[\theta=\arccos\left( \frac{\frac{-1}{4}}{\frac{1}{2}} \right)
    =\arccos\left(  \frac{-1}{2}\right)=120^\circ\]
\end{solution}

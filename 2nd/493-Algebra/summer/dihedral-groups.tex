\subsection{Intro}
\begin{definition}
    A group is a set \(G\) together with a binary operation \(*:G\times G\to G\) s.t.
    \begin{itemize}
        \item \textbf{(Closure)} For all \(a,b\in G\) we have \(a*b\in G\)
        \item \textbf{(Identity)} There is \(e \in G\) s.t. for all \(a \in G\;\; a*e=e*a=a\)
        \item \textbf{(Inverses)} For all \(a \in G\) there is an \(a^{-1}\in G \text{ s.t. } a*a^{-1}=a^{-1}*a=e\)
        \item \textbf{(Associativity)} For all \(a,b,c\in G \text{ we have } a*(b*c)=(a*b)*c\)
    \end{itemize}
\end{definition}

\begin{definition}
    Given a group \(G \text{ and } g\in G\) we say the order of \(g\) is \(n \in\N\) if \(n\) is the minimum value s.t. \(g^n=g*g*\cdots*g=e\) \\
    Write as: ord\(\left( g \right)=\mid g \mid=n\)
\end{definition}

\subsubsection{Exercises}
\begin{example}
    Let \(G =\R\setminus\left\{ -1 \right\} \t{ and define } a * b = a+b+ab \t{. Show } \left( G\t{,}* \right)\) is an Abelian group
    \begin{itemize}
        \item (Abelian) For all \(a,b\in G\;\; a*b=a+b+ab=b+a+ba=b*a\)
        \item (Closure) \checkmark
        \item (Identity) For all \(a \in G\;\; a*0=0*a=0\)
        \item (Inverses) For all \(a \in G\)\s\s let \s\s\(b= \displaystyle \frac{-a}{a+1}\). 
        
        \hspace{11pt} Thus, \(a*b=b*a=a+b+ab=a+b(a+1)=\displaystyle\frac{-a}{a+1}(a+1)=a-a=0\)
        \item (Associativity) For all \(a,b,c \in G\)
        
        \hspace{11pt} \(a*(b*c)=a*(b+c+bc)=a+b+c+bc+ab+ac+abc=(a+b+ab)*c=(a*b)*c\)
    \end{itemize}
    Therefore, \(G\) is an Abelian group
\end{example}

\subsubsection*{Operations}

\begin{align*}
    r \circ S_A(\triangle ABC) &=r(\triangle ACB)=\triangle BAC =S_C(\triangle ABC) & r \circ S_A = S_C\\
    r^2 \circ S_A(\triangle ABC) &=r^2(\triangle ACB)=\triangle CBA = S_B(\triangle ABC) & r^2 \circ S_A = S_B
\end{align*}
Now we only need \(S_A\) so lets call this \(S\)\\
Thus we only need \(S,rs,r^2S,r,r^2,e\)\\
Also \(S^2=r^3=e\)

\subsubsection*{How \(S,r,r^2\) Commute}
\begin{align*}
    r \circ S(\triangle ABC) &=r(\triangle ACB) =\triangle BAC = S\circ r^2\\
    r^2 \circ S(\triangle ABC) &=r^2(\triangle ACB)=\triangle CBA = S \circ r
\end{align*}

\begin{definition}
    The group of symmetries of an equilateral triangle is D\textsubscript{3}\(=\left\{ e,r,r^2,S,Sr,Sr^2 \right\}\), where  \(r^3=S^2=e\), \(rS=Sr^2\)

    Dihedral group of order \(2\cdot3=6\)

    \textbf{\large Compact Notation} \large D\textsubscript{3}\(=\langle r,s\footnotemark \mid  r^3=S^2=e \t{ , } rS=Sr^2 \rangle\)
\end{definition}
\footnotetext{The generators of D\textsubscript{3} meaning you can create all other operations from them}

\subsubsection{Symmetry Group of regular n-gon}
\phantom{text}

Rotation: \(r=\dfrac{360^\circ}{n}\;\;\circlearrowleft \left( e,r,r^2,\ldots,r^{n-1} \right)\)
\vspace{5pt}

Reflection: \(s=\)reflection across middle \(\left( s,sr,sr^2,\ldots, sr^{n-1} \right)\)

\begin{proof}[\textbf{Claim:} \(rs=sr^{n-1}\)]
    \phantom{text}\\
    \textit{Proof.}
    We only need 3 points since our shape is defined by the order of those 3 points

    \(r\circ s\left( n,1,2 \right) = r\left( 2,1,n \right) = \left( 1,n,n-1 \right)\)
    
    \(s\circ r^{n-1}\left( n,1,2 \right) = s\left( n-1,n,1 \right) = \left( 1,n,n-1 \right)\)

    Thus \(rs=sr^{n-1}\)
\end{proof}

\begin{definition}
    The dihedral group D\textsubscript{n} is the symmetry group of a regular n-gon.

    D\textsubscript{n}\(=\langle s,r \mid r^n=s^2=e \t{ , } rs=sr^{n-1} \rangle \)
\end{definition}

\begin{proof}[\textbf{Claim:} For all \(1\leq m\leq n-1\) we have \(r^ms=sr^{n-m}\)]
    \phantom{text}\\
    \textit{Proof.} Base Case: \(m=1 \implies rs=sr^{n-1}\) \checkmark

    Inductive Step: Assume for \(k \geq 1\) the statement is true

    Consider: 
    \begin{align*}
        r^{k+1}s&=rr^ks &\t{Apply indcutive hypothesis on } r^ks\\
                &=rsr^{n-k} \\
                &=sr^{n-1}r^{n-k}\\
                &=sr^nr^{n-(k+1)}\\
                &=sr^{n-(k+1)}
    \end{align*}
\end{proof}

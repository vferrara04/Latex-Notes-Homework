\subsection{Sylow Theorems}


\begin{theorem}
    Let $G$ be a finite group and $p$ be a prime, $p\mid |G|$\\
$n=|G|=p^r\times m, p \nmid m$\\
    $G$ contains a subgroup of order $p^r$.
\end{theorem}
\begin{theorem}
    Any two $p-$Sylow subgroups of $G$ are conjugate in $G$.
\end{theorem}
\begin{theorem}
    The number of $p-$Sylow subgroups
    \begin{enumerate}
        \item divides $m$
        \item is congurent to 1 mod $p$
    \end{enumerate}
\end{theorem}

\begin{example}
    Classify groups of order $15$\\
    $n_3$ divides 5\\
    $\equiv 1$ mod 3 thus $n_3=1$\\
    $n_5$ divides 3\\
    $\equiv 1$ mod 5 thus $n_5=1$\\
\end{example}

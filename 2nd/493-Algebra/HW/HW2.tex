\subsection*{Problem 1}
Let $G$ be a group and $N \unlhd G$\\
Let $\psi$ be an homomorphism form $G$ to $G_1$ s.t. $N \subseteq \text{ ker}(\psi)$\\
Define: $\bar{\psi}: G \diagup N \rightarrow G_1$ s.t. $\bar{\psi}(gN)=\psi(g)$\\\
Well Defined: Suppose $gN=hN \implies gh^{-1} \in N \implies gh^{-1} \in $ ker$(\psi)$\\
$\psi(gh^{-1})=e \implies \psi(g)=\psi(h)$ since $\psi$ is a homomorphism this implies $\bar{\psi}(gN)=\bar{\psi}(hN)$\\
Homomorphism: Let $gN,hN \in G \diagup N$\\
$\bar{\psi}(gNhN)=\bar{\psi}(ghN)=\psi(gh)=\psi(g)\psi(h)=\bar{\psi}(gN)\bar{\psi}(hN)$\\
Uniqueness: Suppose there exists another homomorphism $\phi: G\diagup  N \rightarrow G_1$ such that $\varphi=\psi \circ \varphi$\\
\[
\phi(gN)=\psi(g) \implies \phi = \bar{\psi}
\]

\subsection*{Problem 2}
Let $H$ be a subgroup of $G$.
\begin{itemize}
    \item (Inverses) Let $a \in N_G(H)$\\
    Consider: $a^{-1}Ha=a^{-1}aH=H$ thus $a^{-1} \in N_G(H)$
    \item (Closure) Let $a,b \in N_G(H)$\\
    Consider: $abHb^{-1}a^{-1} = abb^{-1}Ha^{-1}=aHa^{-1}=H$ thus $ab \in N_G(H)$
    \item (Identity) Consider $eHe=H$ thus $e \in N_G(H)$
\end{itemize}
Let $h \in H$\\
Consider : $hHh^{-1}$ since $h,h^{-1} \in H \implies hHh^{-1}=H$ thus $h \in N_G(H)$ thus $H \subseteq N_G(H)$\\
By definition since $\forall g \in N_g(H)\; gHg^{-1}=H$ thus $H \unlhd N_G(H)$\\
Largest: Let $K$ be a subgroup of $G$ s.t. $H \unlhd K$ thus $\forall k \in K$ 
\[kHk^{-1}=H \]
thus $k \in N_G(H)$ thus $K \subseteq N_G(H)$

\subsection*{Problem 3}
\begin{itemize}
    \item (i)
    \begin{proof}
        \phantom{text}
        \begin{itemize}
            \item (Closure) Let $hk,jl \in HK$ s.t. $h,j \in H$ and $k,l \in K$\\
            $hkjl=h(kj)l$ since $H \subseteq N_G(K) \implies kj=jk'$ for $k' \in K$\\
            $h(jk')l=hjk'l \in HK$
            \item (Identity) Since $H,K$ are subgroups of $G$, $e\in H\cap K \implies ee=e\in HK$
            \item (Inverses) Let $hk \in HK$ s.t. $h\in H$ and $k \in K$\\
            Consider $k^{-1}h^{-1}$ since $H \subseteq N_G(K) \implies k^{-1}h^{-1}=h^{-1}k'$ for $k' \in K$\\
            thus $k^{-1}h^{-1}=h^{-1}k'\in HK$\\
            \phantom{text}\\
            \item Let $hk \in HK$ s.t. $h\in H$ and $k,k' \in K$\\
            $(hk)k'(hk)^{-1}=h(kk'k^{-1})h^{-1}$ thus $kk'k^{-1}=l \in K$ and since $h \in H \subseteq N_G(K)$\\ 
            $\implies hlh^{-1} \in K$ thus $(hk)k'(hk)^{-1} \in K$ thus $K \unlhd HK$
        \end{itemize}
    \end{proof}
    \item (ii)
    \begin{proof}
        Let $l \in H \cap K \implies l \in H$ and $l \in K$ and $h \in H$\\
        $hlh^{-1}$ since $l \in K$ and $H \subset N_G(K) \implies hlh^{-1} \in K$\\
        since $h \in H$ and $l \in H \implies hlh^{-1} \in H$ thus $hlh^{-1} \in H \cap K$ thus $H\cap K \unlhd H$
    \end{proof}
    \item (iii)
    \begin{proof}
        Let $\varphi: H \rightarrow HK \diagup K$ s.t. $\varphi(h)=hK$\\
        Well-defined: Suppose $h_1 = h_2 \in H \implies \varphi(h_1)=h_1K=h_2K=\varphi(h_2)$\\
        Homomorphism: Let $h,j \in H$\\
        $\varphi(hj)=hjK=hKjK=\varphi(h)\varphi(j)$\\
        Kernel: Let $h \in \text{ker}(\varphi) \unlhd H \implies h \in H$\\
        $\varphi(h)=hK=K \implies h \in K \implies h \in H \cap K \implies$ ker$(\varphi)=H \cap K$\\
        Surjective: Let $hkK \in HK \diagup K$, for $h \in H,\; k\in K$\\
        $\varphi(h)=hK=hkK$ thus im$(\varphi)=HK \diagup K$\\
        Thus by the first isomorphism theorem $H \diagup H \cap K \cong HK \diagup K$
    \end{proof}

\subsection*{Problem 6}
Let $G$ be order 8.\\
By lagrange's theorem $G$ can only have terms of order 1,2,4,8
\begin{enumerate}
    \item If $G$ has an element $g \in G$ s.t. order of $g$ is 8.\\
    $\ang{g} \leq G$ but since $|\ang{g}|=|G| \implies \ang{g}=G \implies G$ is cyclic thus $G \cong C_8$
    \item If $G$ does not have an element of order 8 then $G$ has elements of order 1,2,4
    By Cauchy's theorem since $2 \mid 8=|G| \implies \exists a \in G$ s.t. ord$(a)=2$ and let $\ang{a}=K \cong C_2$\\
    Consider $G \diagup K$ by Lagrange's theorem $|G \diagup K|=4$ thus proven in class $G \diagup K \cong C_4$ or $G \diagup K \cong C_2 \times C_2$
\end{enumerate}

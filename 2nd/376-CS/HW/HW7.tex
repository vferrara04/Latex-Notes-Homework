\begin{subproblem}{}
    Prove that all efficiently decidable languages are efficiently verifiable. In other words, show that every language in $\mathsf{P}$ has an efficient verifier. 
\end{subproblem}
\begin{problemproof}
    Let $L \in \mathsf{P}$ thus there exists an efficient Turning Machine $M$ that decides $L$ in polynomial time.

    Let $V(x,c)=M(x)$.

    $V(x,c)$ is polynomial time since $M(x)$ is.

    If $x \in L$ then $M(x)$ accepts thus $V(x,c)$ accepts.

    If $x \notin L$ then $M(x)$ rejects thus $V(x,c)$ rejects for all $c$ since we ignore it.
\end{problemproof}

\begin{subproblem}{}
    Prove that every language in $\NP$ is decidable. In other words, show that every language in $\NP$ has a decider. 
\end{subproblem}
\begin{problemproof}
    Let $L \in \NP$ thus there exists an efficient verifier $V$ that is bounded by $p(|x|)$ a polynomial function.

    Define $M(x)=V(x,c)$ s.t. we loop over all strings $c$ s.t. $|c| \leq p(|x|)$ and accept if $V(x,c)$ accepts or reject if $V(x,c)$ rejects for all $c$.

    If $x  \in L$ then there exists a $c$ s.t. $p(|x|)\geq|c|$. Thus, our Turing machine $M(x)$ will loop through all possible strings $c$. Thus, eventually we will find a $c$ s.t. $V(x,c)$ accepts thus $M(x)$ accepts.

    If $x \notin L$ then there is no $c$ s.t. $p(|x|) \geq |c|$ s.t. $V(x,c)$ will accept. Thus, $M(x)$ will reject.

    Thus, $M$ is a decider for $L$. Thus, $\NP$ is a subset of decidable languages.
\end{problemproof}

    \[ 
        \SubsetSum = \{(S, k) : S \text{ is a \emph{multiset} of integers $\geq 0$, and } \exists\; S' \subseteq S \text{ such that } \sum_{s \in S'} s = k\}.

    Prove or disprove: if $A \leq_p B$, then $A \leq_T B$.

   (A disproof would consist of a pair of languages such that $A \leq_p B$ but $A \not \leq_T B$.)
\end{subproblem}
\begin{problemproof}
    Assume $A \leq_p B$ then this means there exists a efficient computable $f : \Sigma^* \to \Sigma^*$ s.t. $x \in A \iff f(x) \in B$.

    Let $M_B$ be an oracle that decides $B$.

    Define: $M_A(x)=M_B(f(x))$.

    $M_A$ halts on any input because $f$ is efficiently computable, and $M_B$ halts on every input.    
    
    Therefore,
    \begin{align*}
        x \in A &\iff f(x) \in B \\
                &\iff M_B(f(x)) \t{ accepts}\\
                &\iff M_A(x) \t{ accepts}
    \end{align*}

    Thus, $M_A$ decides $A$. Thus, $A \leq_T B$.
\end{problemproof}

\newcommand{\ptimemap}{\leq_p}
\begin{subproblem}{}
    Prove that PTMRs are \emph{transitive}: if $A \ptimemap B$ and $B \ptimemap C$, then $A \ptimemap C$.
\end{subproblem}
\begin{problemproof}
    Assume $A \leq_p B$ and $B \leq_p C$.

    Thus, there exists efficiently computable $f,g$ s.t. $x \in A \iff f(x) \in B$ and $y \in B \iff g(y) \in C$.

    Thus, $g \circ f$ is efficient computable since $y=f(x)$ can be computed in $O(|x|^c)$ for some constant $c$. Since writing down is part of the output time we know that $|y|=O(|x|^c)$.

    Thus, $g(y)$ can be computed in $O(|y|^d)$ for some constant $d$.

    Thus, $g\circ f(x)$ can be computed in $O(|x|^c)+O(|f(x)|^d)=O(|x|^c+|x|^{cd})$ which is a polynomial in $|x|$.

    Thus, $x \in A \iff f(x) \in B \iff g(f(x)) \in C$.

    Therefore, $A \leq_p C$
\end{problemproof}

\begin{problem}{Dominating sets.}
    

    Consider an undirected graph $G = (V,E)$. A \emph{dominating set} of a graph is a subset of vertices $D$ such that every vertex in $V$ is either in $D$ or connected to some vertex/vertices in $D$. For example, consider the following graph:

    \begin{center}
        \begin{tikzpicture}[scale=1.5, every node/.style={draw, circle, minimum size=1cm, inner sep=0pt}]
            % Define the nodes for the first graph
            \node (a) at (2,1) {$a$};
            \node (b) at (1,0) {$b$};
            \node (c) at (3,0) {$c$};

            % Draw edges
            \draw (a) -- (b);
            \draw (b) -- (c);
            \draw (c) -- (a);
        \end{tikzpicture}
    \end{center}

    The graph has a dominating set of size $1$, for example, $D = \{a \}$ is a dominating set because $a \in D$ and both $b$ and $c$ are connected to $a$. 

    Notice the difference between a dominating set and a vertex cover: In the graph above, $D = \{a \}$ is \emph{not} a vertex cover because it does not cover the edge $(b,c)$. However, $C = \{b,c\}$ is both a vertex cover and a dominating set. 

    Let $G$ be an undirected graph and $k$ be a nonnegative integer. Consider the following language:
    \[
        \DomSet = \{(G,k) = \text{$G$ has a dominating set of size (at most) $k$}\}.
    \]

    In this problem, you will show that $\DomSet$ is $\NP$-complete.
\end{problem}
\begin{subproblem}{}
    Prove that $\DomSet \in \NP$. Specifically, you need to 
              \begin{enumerate}[(i)]
                \item define a certificate and argue that it is polynomial in size with respect to the instance,
                  \item give an efficient verifier for the language that takes in an instance and a certificate as input, 
                  \item give a correctness argument for the verifier, and
                  \item give a runtime analysis for the verifier.
              \end{enumerate}
              In your verifier, you may assume that the certificate is well-formed, so you do not have to check for it's \emph{data type} (e.g., a graph), but you still have to check for its \emph{validity}.\label{dom-set-np}
\end{subproblem}
\begin{problemproof}
    Given a graph $G=(V,E)$ with $|V|=n$ let our certificate $C$ be a $C[1,\dots,n]$ array of bits.

    The certificate is exactly $n$ bits long which is polynomial in $|G|>|V|$

    Let $V$ be a verifier s.t. $(G,k)$ and $C$ is its input.
    \begin{enumerate}
        \item Count:
        \begin{itemize}
            \item Counter the number of $1$'s in $C$.
            \item If the count is greater then $k$, then reject
        \end{itemize}
        \item Check:
        \begin{itemize}
            \item For each $v_i \in V$. Assume we have an order already build into the representation of $V$.
            \begin{itemize}
                \item If $C[i]=1$ then $v_i$ is in the dominating set, so it is dominated.
                \item If $C[i]=0$ then check if there exists a neighbor $v_j$ of $v_i$ s.t. $C[j]=1$
                
                If there is no such neighbor rejct.
            \end{itemize}
        \end{itemize}
        \item Finally Accept.
    \end{enumerate}

    Correcteness:

    If the verifier accpets, then every vertex $v$ is either in the set or has a neighbor $v_j$ with $C[j]=1$. Thus, the set of vertices corresponding to 1's in $C$ is a dominating set of $G$ and its size is at most $k$.

    If $G$ has a dominating set of $D$ of size at most $k$, then one can construct a certificate $C$ by setting $C[i]=1$ if $v_i \in D$ and $C[i]=0$ if $v_i \notin 0$

    Runtime:

    First we count the $n$-bit array which takes $O(n)=O(|V|)$.

    Then for each vertex $v \in V$, we check its neighbors. If the degree of $v$ is $d_v$, then the total time for all vertices is $O(\sum_{v \in V} d_v)=O(2|E|)=O(|E|)$.

    Thus, the total running time is $O(|V|+|E|)$ which is polynomial in the size of the input graph.
\end{problemproof}

    If $\tau(x_i)=false$ then choose the $F_i$

    In each gadget, the chosen vertex either $T_i,F_i$ is adjacent to the other vertex in the pair $T_i,F_i$ and to both $\alpha_i,\beta_i$.

    If $\overline{x_i}$ is in $C_j$ and $\tau(x_i)=false$, then $C_j$ is adjacent to $F_i \in D$.

    Thus, every $C_j$ is adjacent to some vertex in $D$.

    Thus, $D$ is a dominating set of size $k$.

    Let $(G,k) \in \DomSet$ s.t. it has the structure defined by $f$.

Thus, given that $|\phi|=O(n+m)$ where $n$ is the number of variables and $m$ is the number of clauses. The total number of edges and vertices is $O(n+m)$. \end{problemproof}

\begin{problem}{Subgraph isomorphism.}
    For two graphs $G = (V,E)$ and $H = (W, F)$, an \emph{isomorphism} between them is a \emph{bijection} $f:V \to W$ such that $(u,v) \in E$ if and only if $(f(u), f(v)) \in F$. Two graphs are said to be \emph{isomorphic} if there exists an isomorphism between them. For example, the following two graphs are isomorphic, by the bijection $f$ where $f(1) = 1$, $f(2) = 3$, $f(3) = 5$, $f(4) = 2$, and $f(5) = 4$.

     Recall that a subgraph $G' = (V', E')$ of $G = (V, E)$ is a graph such that $V' \subseteq V$ and $E' \subseteq E$. Let $G$ and $H$ be undirected graphs. Consider the following language
      
      \begin{align*}
        \subgraphiso &= \\ \set{(\ang{G}, \ang{H}): &\text{$G$ has a subgraph $G'$ that is isomorphic to $H$}}.
      \end{align*}
      In this problem, we will prove that $\subgraphiso$ is $\NP$-complete.
\end{problem}

\begin{subproblem}{}
    Prove that $\subgraphiso \in \NP$.
\end{subproblem}
\begin{problemproof}
    Let the certificate $C$ be an array of vertices of $V$ with length $|W|$.

    This is polynomial with respect to $(\ang{G},\ang{H})$ since it is just a subset of the vertices.

    Let $V$ be a verifier s.t. $(\ang{G},\ang{H})$ and $\{(\ang{w},\ang{v})\}$ as input.

    \begin{enumerate}
        \item Check injection:
        \begin{itemize}
            \item Check that for every $C[i]$ and $C[j]$ s.t. $i \neq j$ that $C[i] \neq C[j]$
            \item If $C[i]=C[j]$ for $i \neq j$ then reject.
        \end{itemize}
        \item Check edges
        \begin{itemize}
            \item For every edge $(w_i,w_j) \in F$ check that $(C[i],C[j]) \in E$
            \item If not reject
        \end{itemize}
        Then accept.
    \end{enumerate}

    Correctness:

    If the verifier accepts, then the ordered pairs can be formed into a function $f$ s.t. $f(w_i)=C[i]$. In step one we check it was an injection, and in step two we check it perserved edges thus $f(W) \cong H$, and so $\{\ang{G},\ang{H}\} \in \subgraphiso$

    If there exists a subgraph $G'$ that is isomorphic to $H$ then we know there is an isomorphism form $W \to G'$. Let $C[i]=f(w_i)$. Thus, since $f$ was an injection we know it will pass check 1, and since it is an isomorphism we know it will check step 2. Thus, $V$ will accept the certificate $C$.

    Runtime:

    Step 1 will take $O(|W|^2)$ time to check all pairs in the $C$ array.

    Step 2 will take $O(|F|)$ steps to verify the edge-preservation.

    Since both $|W|$ and $|F|$ are bounded by the input we know the overall verification runs in polynomial time with respect to the input.
\end{problemproof}

\begin{subproblem}{}
    Prove that $CLIQUE \ptimemap \subgraphiso$. Specifically, you need to 
            \begin{enumerate}[(i)]
                \item give a PTMR from $CLIQUE$ to $\subgraphiso$,
                \item give a correctness argument for the reduction, and
                \item give a runtime analysis for the reduction.
            \end{enumerate}
            Since $CLIQUE$ is $\NP$-hard, we can conclude that $\subgraphiso$ is $\NP$-hard. 
\end{subproblem}
\begin{problemproof}
    Define $R : \sigma^* \to \sigma^*$ s.t.

    $R(\ang{G},\ang{k})=(\ang{G},\ang{K_k})$

    To perform this algorithm all you have to do is construct $K_k$ which takes $O(k^2)$ time since we create $k$ vertices and $\binom{k}{2}$ edges.

    Then we copy $G$ which takes $O(|G|)$.

    If $G$ has a clique of size $k$, then there exists a set $V' \subset V$ with $|V'|$ s.t. the subgraph induces by $V'$ is isomorphic to $K_k$ thus $(G,K_k) \in \subgraphiso$
    
    If $(G,K_k) \in \subgraphiso$, then there is a subgraph $G'$ of $G$ that is isomorphic to $K_k$. By definition, this means that $G'$ is a clique of size $k$, so $(G,k) \in CLIQUE$

    Runtime:

    Like I said above it takes $O(|G|+k^2)$ times to copy over $G$ and create $K_k$ thus this is polynomial of the size of the input $|(G,k)| \approx |G| + |k|$.

    Therefore, $CLIQUE \ptimemap \subgraphiso$
\end{problemproof}
\newcommand{\IS}{\textsc{IS}}
\begin{problem}{ig independent sets.}
    Recall that an \emph{independent set} of an undirected graph $G=(V,E)$ is a subset $I \subseteq V$ such that no two (distinct) vertices in~$I$ have an edge between them.

  In lecture, we proved that the following language is $\NP$-complete.
  \[
    \IS = \set{(G,k) : G = (V,E) \text{ has an independent set of size (at least) } k} .
  \]

  In this problem, we will explore the following variant of $\IS$. 

    \newcommand{\BigIS}{\textsc{BIG-IS}}
    \[
      \BigIS = \set{ G : G=(V,E) \text{ has an independent set of size (at least) } |V|-1 } .
    \]
  
    Either prove that $\BigIS\in \P$ (by giving a polynomial-time algorithm that decides $\BigIS$) or that $\BigIS$ is $\NP$-hard (by showing that $\IS \ptimemap \BigIS$).
\end{problem}
\begin{problemproof}
    This checks to see if there is an independent set of size $|V|-1$
    \begin{algorithm}[H]
    \caption{BIG-IS}
    \begin{algorithmic}[1]
        \Require An undirected graph $G=(V,E)$.
        \Ensure \textbf{Accept} if $G$ has an independent set of size $|V|-1$, \textbf{Reject} otherwise.
        \If{$|E|=0$}  \State \Return \textbf{Accept} \EndIf
        \For{each vertex $v \in V$}
            \State $valid \gets \text{true}$
            \For{each edge $(u,w) \in E$}
                \If{$u \neq v$ \textbf{and} $w \neq v$}
                    \State $valid \gets \text{false}$
                    \State \textbf{break} from inner loop
                \EndIf
            \EndFor
            \If{$valid$ is true}
                \State \Return \textbf{Accept} \Comment{$V\setminus\{v\}$ is an independent set of size $|V|-1$}
            \EndIf
        \EndFor
        \State \Return \textbf{Reject}
        \end{algorithmic}
        \end{algorithm}

    Correctness:

    This halts on every input since we do two for loops but since $G$ is finite we will end.

    If $G$ is totally disconnected then $G$ trivially has an independent set $I$ of size at least $|V|-1$, so this implies that $|E|=0$ thus we accept on line 2.
    
    If the algorithm accepts on line two then this implies that $|E|=0$ thus $G$ is totally disconnected and trivially has an independent set $I$ of size at least $|V|-1$.

    If $G$ is not totally disconnected and has an independent set $I$ of size $|V|-1$.

    Let $v \in V \setminus I$

    Assume for contradiction that there exists an edge $(u,w) \in E$ s.t. both $u,w \in I$ and $u\neq v\neq w$. Since $I$ is an independent set, no two distinct vertices in $I$ can be adjacent, and so such an edge cannot exist.

    Thus, when we are on the loop for vertex $v$, we will never hit step 7 in our algorithm thus $valid$ will always be true. Thus, we will accept $G$.

    If our algorithm accepts on line 10 then this means that for some $v \in V$ every edge $(u,w) \in E$ there was no paring s.t. $u=v$ or $v=w$. This implies that no two vertices in $V \setminus \{v\}$ have an edge between them. Thus, $I=V \setminus \{v\}$ is an independent set.

    Thus, our algorithm decides \textsc{BIG-IS}

    Runtime:

    The algorithm loops $|V|$ times then it loops $|E|$ times. The rest of the operations are $O(1)$.

    Thus overall we take $O(|V||E|)$ time which is a polynomial on the input.
\end{problemproof}

\newcommand{\coNP}{\mathsf{coNP}}
\begin{problem}{A ticket to \$1 million (optional extra credit)}
    The complexity class $\coNP$ is defined as the set of languages whose \emph{complement} languages are in $\NP$:
  \[
    \coNP = \set{\overline{L} : L\in\NP}
  \]
  Note that $\coNP \neq \overline{\NP}$, which is the class of languages that are not in $\NP$. 

In this problem, we will explore the complexity class \emph{difference polynomial-time}, denoted as $\DP$, which contains all languages that can be expressed as the intersection of an $\NP$ language and a $\coNP$ language:
  \[
    \DP = \set{L: \text{$L = A \cap B$ for some $A \in \NP$ and $B \in \coNP$}},
  \]
      or equivalently, 
      \[
        \DP = \set{L: \text{$L = A \cap \overline B$ for some $A,B \in \NP$}}.
      \]

  Note that $\DP \neq \NP \cap \coNP$, which is the set of languages that are both in $\NP$ and in $\coNP$.
\end{problem}
\newcommand{\TSTUS}{\textsc{3SAT-3UNSAT}}
\begin{subproblem}{}
    Let $\phi_1$ and $\phi_2$ be 3CNF formulas. Consider the following language
    
    \[
        \TSTUS = \set{(\phi_1, \phi_2): \text{$\phi_1$ is satisfiable but $\phi_2$ is not}}.
    \]
    Show that $\TSTUS$ is $\DP$-complete. In particular, 
      \begin{enumerate}[(i)]
          \item briefly explain why $\TSTUS \in \DP$, and
          \item prove that $\TSTUS$ is $\DP$-hard, i.e., $L \ptimemap \TSTUS$ for all $L \in \DP$.
        \end{enumerate}
\end{subproblem}
\begin{problemproof}
    \begin{itemize}
        \item $L_1=\{(\phi_1,\phi_2) \mid \phi_1 \t{ is satisfiable}\}=\TSAT \in \NP$
        This is in $\NP$ because determining if a given 3CNF formula is satisfiable is exactly $\TSAT$
        \item $L_2=\{(\phi_1,\phi_2) \mid \phi_2 \t{ is unsatisfiable}\}=\overline{\TSAT} \in \coNP$
        This is in $\coNP$ because determining if a given 3CNF formula is unsatisfiable is exactly the oposite of checking if it is satisfiable thus it is the complement of $\TSAT$.

        Thus, $\TSTUS=L_1 \cap L_2 \in \DP$
        
        \item Let $L=A \cap \overline{B}$ for some $A,B \in \NP$.
        
        Since $A,B \in \NP$ and $\TSAT$ is $\NP$-hard. We know there are efficiently computable functions $f,g$ s.t.
        \[x \in A \iff f(x) \in \TSAT\]
        \[x \in B \iff g(x) \in \TSAT\]

        Define $R(x)=(f(x),g(x))$

        Correctness:
        \begin{align*}
            x \in L &\iff x \in A \t{ and } x \notin B\\
                    &\iff f(x) \in \TSAT \t{ and } g(x) \notin \TSAT\\
                    &\iff (f(x),g(x)) \in \TSTUS
        \end{align*}

        Runetime:

        Since $f,g$ are efficient computable this means $f(x)$ runs in $O(|x|^d)$ time and $g(x)$ runs in $O(|x|^c)$ time. Thus, $R(x)$ runs in $O(|x|^d+|x|^c)$. Thus, $R$ is efficiently computable. Thereforem, $\TSTUS$ is $\DP$-complete.
    \end{itemize}
\end{problemproof}

\begin{subproblem}{}
    Show that if $\DP \neq \NP$, then $\P \neq \NP$.
\end{subproblem}
\begin{problemproof}
    Assume $\DP \neq \NP$. Assume for contradiction that $\P=\NP$.

    Since $\NP=\P$ is closed under complement we have $\coNP=\P$.

    Thus, $\NP=\P=\coNP$.

    Let $L \in \DP$. Thus, $L=A \cap \overline{B}$ for $A,B \in \NP$. Since $\NP=\P=\coNP$ and that $\P$ is closed under intersection this implies that $L \in \P$.

    Thus, $\DP \subset \P=\NP$

    Let $L \in \NP=\coNP$ consider $L=L \cap L$. Thus, $L \in \DP$.

    Therefore, $\DP = \NP$ which is a contradiction. Thus, $\P \neq \NP$.
\end{problemproof}

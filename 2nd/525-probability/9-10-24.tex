\subsubsection*{Random Variabiles}
\begin{definition}
    Given $(\Omega, F, \P)$, a random variable $X$ is a function $X: \Omega \rightarrow \R$
    such that $\{X \leq a\}=\{\omega \in \Omega \mid X(\omega) \leq a\} \in F$ for any $a \in R$
\end{definition}

\subsubsection*{What we talk about when we talk about a random variable}
(Cumulative) distribution function
\[
F_X(a)=\P(X \leq a)
\]
(Discrete) probability mass function
\[
f_X(a)=\P(X=a)
\]

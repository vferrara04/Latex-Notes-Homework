\section{Basis for a topology}
\begin{definition}{Basis}{}
    Let $X$ be a set. A collection $\beta$ of subsets of $X$ is
    a basis for a topology if:
    \begin{itemize}
        \item every $x\in X$ is in some $B \in \beta$
        \item If $x \in B_1 \cap B_2$ for $B_1,B_2 \in \beta$\\
        then: $\exists B_3 \in B$ s.t. $x \in B_3 \subseteq B_1 \cap B_2$
    \end{itemize}
    The topology generated by $\beta$ is the collection $T$ if subsets defined by:
    \[U \in T \iff \t{ for every } x\in U, \exists B \in \beta \t{ s.t } x\in B \subseteq U\]
\end{definition}
\begin{example*}{}{}
    $X = \R$, $\beta=$ set of open intervals in $\R$\\
    $X = \R^n$, $\beta=$ set of open balls $B(x,r)=\{y \in \R^n \mid d(x,y) < r\}$\\
    Check that the second example is actually a basis for a topology
    \begin{itemize}
        \item every $x \in \R^n$ lies in some open ball. $x \in B(x,1)$
        \item Assume $x \in B_1 \cap B_2=B(y_1,r_1) \cap B_(y_2,r_2)$\\
        Set $r_3=\min(r_1-d(x,y_1),r_2-d(x,y_2))$\\
        Consider the ball $B_3=B(x,r_3)$. Let $z \in B_3$.\\
        $d(z,y_1) \leq d(x,y_1) + d(z,x) < d(x,y_1) + r_3 \leq d(x,y_1) + r_1 - d(x,y_1)=r_1$\\
        So $z \in B(y_1,r_1) = B_1$ do the same proof for $B_2$ thus $z \in B_1 \cap B_2$ thus $B_3 \subseteq B_1 \cap B_2$
    \end{itemize}
    \textcolor{exmscol!50!white}{\hbox{}\nobreak\hfill$\blacksquare$}
\end{example*}
\begin{proposition}{}{}
    This is actually a topology on $X$.
\end{proposition}
\begin{proppf}
    \begin{itemize}
        \item $\emptyset \in T$ vacuously, and $X \in T$ comes from first basis axiom.
        \item Suppose $\{u_j\}_{j\in J}$ of sets in $T$.\\
        Let $x \in \bigcup_{j \in J} u_j$. Thus $x\in u_{j'}$ for some $j' \in J$, so $\exists B_{j'} \in \beta$ s.t. $x \in B_{j'} \subseteq u_{j'}$
        thus $x \in B_{j'} \subseteq \bigcup_{j \in J} u_j$. This shows $\bigcup_{j \in J} u_j \in T$
        \item Suppose $x \in u_1 \cap u_2$ for $u_1,u_2 \in T$. Then $x \in B_1,B_2$ for sets $B_1,B_2 \in \beta$ s.t. $B_1 \subseteq u_1$ and $B_2 \subseteq u_2$ thus $x \in B_1 \cap B_2$. By the second axiom thus implies
        $\exists B_3 \in \beta$. Thus $x \in B_3 \subseteq B_1 \cap B_2 \subseteq u_1 \cap u_2 \in T$
    \end{itemize}
\end{proppf}
\rmkb{Every topology is a basis for itself.}

\begin{proposition}{}{}
    Let $X$ be a set, let $\beta$ be a basis for topology $T$. The sets in $T$ are precisely the sets of the form
    \[\bigcup_{j \in J} B_j\] for $\{B_j\}_{j \in J}$ collection of sets in $\beta$
\end{proposition}
\begin{proppf}
    \begin{itemize}
        \item Every set of this form is in $T$:
        \[x \in \bigcup_{j \in J}B_j, \t{ then } x\in B_{j'} \subset \bigcup_{j \in J} B_j\]
        \item Every set in $T$ has the form:
        \[\t{Let } U \in T. \t{ Then for every } x \in U, \t{ there is a } B_x \in \beta \t{ s.t. } B_x \subset U \implies U=\bigcup_{x \in U} B_x\]
    \end{itemize}
\end{proppf}
\noindent Decomposition is NOT in general unique.\\
Basis sets are always open.\\
\subsection{Comparing Topologies}
\begin{definition}{Finer}{}
    If $T_1,T_2$ are topologies on the same set $X$, we say that $T_1$, is finer than $T_2$ if $T_1 \supseteq T_2$. Also, $T_2$ is coarser than $T_1$
\end{definition}
\begin{proposition}{}{}
    Let $\beta,\beta'$ be bases for topologies $T,T'$ on $X$. Then $T \subseteq T'$ ($T'$ is finer than $T$) if and
    only if every $x \in X$, and every $B \in \beta$ containing $x$, $\exists$ some $B' \in \beta'$ s.t. $x \in B' \subset B$
\end{proposition}
\begin{example*}{}{}
    $X = \R^n$, let $\beta$ be the set of open boxes in $\R^n$: sets of the form $(a_1,b_1) \times (a_2,b_2) \times \dots \times (a_n,b_n)$\\
    This is a basis. It generates the same standard topology on $\R^n$.
\end{example*}
\begin{example*}{}{}
    Let $X = \R$. Consider collections of half-open intervals in $\R$: $[a,b),(a,b]$.\\
    Claim: this is a basis for a topology.\\
    This is not the same as the std. topology on $\R$\\
    This is finer than the std. topology.
\end{example*}

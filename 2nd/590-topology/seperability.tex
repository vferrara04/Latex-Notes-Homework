\section{Separation Axioms}

\defn{}{Let $X$ be a topological space
\begin{itemize}
    \item $X$ is $T_1$ if singletons are closed.
    \item $X$ is Hausdorff ($T_2$) if, for every pair of distinct points $x,y \in X$, $\exists$ disjoint neighborhoods of $x,y$
    \item $X$ is regular ($T_3$) if it is $T_1$ and for every $x \in X$ and every closed $B \subset X$ with $x \notin B$, $\exists$ disjoint open sets containing $x$ and $B$.
    \item $X$ is normal if it is $T_1$ and for every pair of disjoint closed subsets $A,B \subset X$, $\exists$ disjoint open sets containing $A,B$.
\end{itemize}
}

\thm{}{
    A subspace of a $T_1,T_2,T_3$ space is $T_1,T_2,T_3$
}

\thm{}{
    Let $\{X_\alpha\}_{\alpha \in J}$ be a family of $T_1,T_2,T_3$ spaces. Then the product is $T_1,T_2,T_3$.
}

\lemp{}{
    Let $X$ be a top. space. Then $X$ is regular $\iff$ for every $x \in X$ and any neighborhood $U$ of $X$, there exists a neighborhood $V$ of $x$ s.t. $\overline{V} \subset U$.
}{
    $(\implies)$ Suppose $X$ is regular. Fix $x \in X$, neighborhood $U$ of $x$.

    Since $X \setminus U$ is closed, so can find disjoint open sets containing $x$, $X\setminus U$. 
    
    $X \setminus U \subset U$ and $x \in V$.

    $X \setminus W$ is closed set containing $V$, so $\overline{V} \subset X \setminus U$, and since $X \setminus U \subset W$, we have $X \setminus W \subset U$, so $\overline{V} \subset U$.

    $(\impliedby)$

    Let $x \in X$, $B \subset X$ s.t. $x \notin B$, assume $B$ is closed.

    Consider $U=X \setminus B$ is a neighborhood of $x$. Let $V$ be a neighborhood of $x$ s.t. $\overline{V} \subset U$.

    Thus, $V,X\setminus \overline{V}$ are open sets separating $x$ and $B$.
}

\thmp{}{
    If $X$ is a metrizable space, $X$ is normal.
}{
    Let $A,B$ be disjoint closed sets subsets of X.

    For each $a \in A$, there exists an $\epsilon_a > 0$ s.t. $B(a,\epsilon_a) \cap B = \emptyset$.

    For each $b \in B$, there exists an $\epsilon_b > 0$ s.t. $B(b,\epsilon_b) \cap A = \emptyset$.

    Consider $B(a,\epsilon_a/2),B(b,\epsilon_b/2)$

    If $z \in B(a,\epsilon_a/2)\cap B(b,\epsilon_b/2)$, then $d(a,b)\leq (\epsilon_a + \epsilon_b)/2 \leq \max(\epsilon_a,\epsilon_b)$. Which is a contradiction.

    Thus,\[\bigcup_{a \in A}B(a,\epsilon_a/2) \quad \bigcup_{b \in B}B(b,\epsilon_b/2)\]
    are disjoint open sets containing $A,B$.
}

\corp{}{
    Manifolds are regular.
}{
    Let $M$ be a $n$-manifold. By definition, $M$ is Hausdorff.

    $M$ is locally homeomorphic to $\R^n$, and $\R^n$ is locally compact, so $M$ is locally compact.

    Thus, $M$ has a 1-point compactification. $\hat{M}$.

    $\hat{M}$ is a compact Hausdorff space. Thus, $\hat{M}$ is normal. Thus, regular, and subspaces of regular spaces are regular.
}

\thmp{}{
    Every second countable regular space is normal.
}{
    Let $\beta$ be a countable basis for $X$.

    Let $A,B$ be disjoint closed subsets of $X$.

    Fix $a \in A$. Then there is a neighborhood $U$ of $a$ that is disjoint from $B$.

    By regularity there is a neighborhood $V$ of $a$ s.t. $\overline{V} \subset U$.

    Thus, there is a $W \in \beta$ s.t. $a \in W \subset U$.

    Then do this for any point $a \in A$, and thus we get a countable open covering of $A$ by sets whose closures are disjoint from $B$.

    Do the same for $B$.

    Call them $\{U_n\}$ and $\{V_n\}$.

    Then for all $n$ define:

    $U_n'=U_n \setminus \bigcup_{i=1}^n \overline{V_n}$

    $V_n'=V_n \setminus \bigcup_{i=1}^n \overline{U_n}$

    Thus, thus union is are disjoint open sets that contain $A,B$.
}

\thm{Urysohn lemma}{
    Let $X$ be a normal space, let $A,B$ be closed subsets. Then there exists a continuous function.
    $f: X \to [0,1]$ s.t. $f(x)=0$ for all $x \in A$, $f(x)=1$ for all $x \in B$.
}

\thmp{Urysohn Metrization Theorem}{
    Every regular space with a countable basis is metrizable
}{
    By Theorem 12.8 we know that this space is normal.

    Let $X$ be normal and second countable.

    \clmp{}{
        There exists countable collection of continuous functions $f_n : X \to [0,1]$ s.t. for
        every point $x \in X$, and every neighborhood $U$ of $x$. $\exists n$ s.t. $f_n(x) = 1$, $f_n(X \setminus U)=\{0\}$
    }{
        \setlength{\parindent}{0pt}
        Let $\beta$ be a countable basis.

        For each pair of indices $n,m$ s.t. $\overline{B_n} \subset B_m$, define function
        $f_{n,m}: X \to [0,1]$ s.t. $f_{n,m}(\overline{B_n})=\{1\}$, $f_{n,m}(X \setminus B_n)=\{0\}$

        The $f_{n,m}$ are indexed by a subset of $\N \times \N$, so this is a countable collection.

        Fix $x \in X$, neighborhood $U$ of $x$, $\exists B_n$ s.t. $x \in B_n \subset U$

        By regularity, $\exists$ neighborhood $V$ of $x$ s.t. $\overline{V} \subset B_n$.

        Thus, $\exists B_m$ s.t. $x \in B_m \subset V$. Thus, $x \in B_m$ and $\overline{B_m} \subset B_n$.

        Thus, $f_{n,m}(x)=1$, $f_{n,m}(X \setminus U) \subset f_{n,m}(X \setminus B_m)=\{0\}$.

        Reindex to the collection $\{f_n\}$
    }

    Define $F: X \to \R^\omega$ s.t. $F(x)=(f_1(x),f_2(x),f_3(x),\dots)$

    $F$ is continuous since each component is continuous.

    Injective:

    Let $x,y \in X$ s.t. $x \neq y$. $X$ is Hausdorff, so there exists a neighborhood $U$ of $x$ s.t. $y \notin U$. Thus, there exists $n$ s.t. $f_n(x) > 0$ and $f_n(z)=0$ for all $z \notin U$, so $f_n(y)=0 \implies f_n(x) \neq f_n(y) \implies F(x) \neq F(y)$.

    Let $Z=F(X)$.

    Let $U$ be open in $X$. Consider $F(U)$.

    Let $z \in F(U)$ thus there exists $x \in U$ s.t. $F(x)=z$. Thus, there is a $n$ s.t. $f_n(x) > 0$ and $f_n(X \setminus U)=\{0\}$.

    Consider the index-n projection map $\pi_n: \R^\omega \to \R$.

    Consider the set $V=\pi^{-1}_n((0,\infty)) \subset \R^\infty$.

    $f_n(x) > 0 \implies f(x)=z \in V$.

    Let $y \in V \cap Z$ then $y=F(c)$ for some $c \in X$, so $f_n(c) > 0$. But then $c \in U$, since $f_n(X \setminus U)=0$.

    Thus, $V \cap Z \subset F(U)$, so $F(U)$ is open. Thus, $F$ is an embedding into $\R^\omega$. Thus, $X$ is homeomorphic to a subspace of a metric space, so $X$ is a metric space.
}

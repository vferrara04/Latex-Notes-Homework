\section{Product Spaces (revisited)}
Let $\{X_\alpha\}_{\alpha \in J}$ be a collection of sets. Can form Cartesian product $\displaystyle\prod_{\alpha \in J} X_\alpha$.

If $X_\alpha=X$, denoted $X^J \iff$ map: $J \to X$

If each $X_\alpha$ top spaces, first gyess for topology on $\displaystyle\prod_{\alpha \in J}X_\alpha$

\defn{Box topology}{
    Topology generated by the basis $\displaystyle\prod_{\alpha \in J}U_\alpha$ s.t. each $U_\alpha$ open in $X_\alpha$.
}

We have natural maps associated to product:
\[\pi_\alpha: \prod_{\alpha \in J} X_\alpha \to X_\alpha\] for each $\alpha \in J$ called projection maps.

\defn{Product Topology}{
    If $\{X_\alpha\}_{\alpha \in J}$ family of top. spaces, product topology on $\displaystyle\prod_{\alpha \in J}X_\alpha$ is the coarsest topology s.t. all projections are continuous.
}
This is in general {\bf NOT} the same as box topology.

A subbasis for product topology is collection of all sets of the form $\pi_\alpha^{-1}(U_\alpha)$ for $U_\alpha$ open in $X_\alpha$.

Each $\pi_{\alpha}^{-1}(U_\alpha)$ is a set of the form $\displaystyle\prod_{\beta \in J}U_\beta$, where all $U_\beta$ for $\alpha \neq \beta$ are $X_\beta$.

Intersection of finitely many elements in subbasis looks like $\displaystyle\prod_{\alpha \in J}U_\alpha$,
where $U_\alpha=X_\alpha$ for all but finitely many $\alpha$.

Sets of this form are a basis for product topology. No way to write a set of the form $\displaystyle\prod_{\alpha \in X}U_\alpha$ where $U_\alpha \subsetneq X_\alpha$ for all $\alpha$ as a union of these basis elements.
This implies box topology is finer than product topology.

Let $\{X_\alpha\}_{\alpha \in J}$ be a family of spaces, for each $\alpha \in J$, let $f_\alpha: Y \to X_\alpha$ be a map for some spaces. Defines a map $f:Y \to \displaystyle\prod_{\alpha \in J}X_\alpha$ by $f(y)=(f_\alpha(y))_{\alpha \in J}$

\propp{
    $f$ is continuous with respect to product topology on $\displaystyle \prod_{\alpha \in J}X_\alpha \iff$ every $f_\alpha$ is continuous.
}{
    $\implies$: By definition of product topology $\pi_\alpha$ is continuous, and $f_\alpha = \pi_\alpha \circ f$, so this is continuous.

    \noindent$\impliedby$: Suffices to prove that $f^{-1}(U)$ is open for every $U$ of the form $\pi_\alpha^{-1}(U_\alpha)$ for some $\alpha \in J$, $U_\alpha \subseteq X_\alpha$ open.

    \[f^{-1}(U)=f^{-1}(\pi_\alpha^{-1}(U_\alpha))=f_\alpha^{-1}(U_\alpha)\]
    open since $f_\alpha$ is continuous.
}

\subsection{Topologies on $\R^J$, where $J$ is infinte}
\prop{
    Let $(X,d)$ be a metric spaces. The function $\bar{d}: X \times X \to \R_{\geq 0}$ by $\bar{d}(x,y)=\min(d(x,y),1)$ is a metric on $X$, induces same topology as $d$.
}

\defn{Diameter}{
    The Diameter of a metric space $(X,d)$ is $\sup\{d(x,y)\}$ for all $x,y \in X$
}

\defn{Uniform Metric}{
    Let $J$ be an index set. Define metric $\bar{\rho}$ on $\R^J$ by
    \[\bar{\rho}(x,y)=\sup_{\alpha \in J}\;\bar{d}(x_\alpha, y_\alpha)\]
}
Induces uniform topology.

When $J$ is finite this is the same topology as product topology on $\R^n$

Otherwise: Uniform Topology lies strictly between the product and box topologies on $\R^J$.

\subsubsection{Product Topology on $\R^\omega$}
\thm{}{
    The function $D:\R^\omega \times \R^\omega \to \R_{\geq 0}$ given by
    \[D(x,y)=\sup_{i \in \N} \dfrac{\bar{d}(x_i,y_i)}{i}\]
    is a metric on $\R^\omega$ which induces product topology.
}

Fix ball $B_D(x,r)$. Choose $N \in \N$ s.t. $1/N < r$. Then $B_{\bar{d}}(x_1,r) \times B_{\bar{d}}(x_2,r)\times,\dots,B_{\bar{d}}(x_N,r)\times \R \times \R \times \dots$
is open in product topology. It's contained in $B_D(x,r)$, since each $y$ in the product satisifies:
\[\bar{d}(x_i, y_i)/i \leq \bar{d}(x_i,y_i)<r \text{ if } i \leq N\]
\[\bar{d}(x_i,y_i) \leq 1/N < r \text{ if } i >N\]

\section{Examples of topological spaces}

\begin{example}{\textbf{Order Topology}}{}
    Let $X$ be a set with the order relation $<$. For $a<b$ in $X$, define intervals:
    \[(a,b)=\{x \in X \mid a < x < b\}\]
    \[[a,b)=\{x \in X \mid a \leq x < b\}\]
    Similar for $(a,b]$,$[a,b]$\\
    Let $\beta$ be the set of all intervals in $X$ of the forms:
    \begin{itemize}
        \item $(a,b)$ for $a<b$ in $X$.
        \item $[a_0,b)$ where $a_0$ is the smallest element of $X$.(if it exists).
        \item $(a,b_0]$ where $b_0$ is the largest element of $X$.(if it exists).
    \end{itemize}
\end{example}
\begin{proposition}{}{}
    This is a basis for a topology on $X$.\\
    This is called the order topology on $X$.
\end{proposition}
\begin{example*}{}{}
    Standard topology on $\R$ is order topology, on $\R$
\end{example*}

\begin{example}{\textbf Product Topology}{}
    Let $X,Y$ be topological spaces.\\
    Let $\beta$ be a collection of sets:
    \[\{U \times V \mid U \t{ open in } X\t{, } V \t{ open in } Y\}\]
    If $\beta_1$ is a basis for topology on $X$, and $\beta_2$ is a basis for topology on $Y$, then 
    \[\{B \times C \mid B \in \beta_1,\,\, C\in \beta_2\}\]
    is a basis for the product topology.
\end{example}
\begin{proposition}{}{}
    This is a basis for a topology.\\
    This is called the product topology on $X \times Y$
\end{proposition}
\begin{example*}{}{}
    The product topology is std topology on $\R$
\end{example*}
\noindent This definition is valid for products of pairs of sets.
\begin{lemma}{}{}
    Suppose $X$ is a topological space. Let $C$ be a collection of open subsets satisfying:
    \begin{itemize}
        \item for each open set $U \subseteq X$ and each $x \in U$, $\exists c \in C$ s.t. $x \in c \subseteq U$
    \end{itemize}
    Then, $C$ is a basis for top. on $X$.
\end{lemma}
\begin{lempf}
    Check $C$ is a basis.
    \begin{itemize}
        \item Every $x \in X$ in some $c \in C$. $X$ is open, so there exists $c$ s.t. $x \in c \subseteq X$
        \item If $c_1,c_2 \in C$, then for any $x \in c_1 \cap c_2$, $\exists c_3 \in C$ s.t. $x \in c_3 \subset c_1 \cap c_2$.\\
        Follows from the fact that $c_1,c_2$ are open which implies $c_1 \cap c_2$ is open so can find $c_3$
    \end{itemize}
    Check: $C$ generates same topology $T$ on $X$.\\
    First, suppose $U$ is open in $T$. By assumption for all $x \in U$, there exists $c \in C$ s.t. $x \in c \subset U$,
    so $U$ is open in the topology generated by $C$.\\
    If $U$ is open in topology generated by $C$. Then $U$ is a union of sets in $C$. But every $c \in C$ is open in $T$, so union $U$ is also open in $T$.
\end{lempf}
\begin{definition}{Metric}{}
    Let $X$ be a set. A function $d: X \times X \to \R$ is a metric if it satisfies:
    \begin{itemize}
        \item $d(x,y)=0 \iff x=y$
        \item $d(x,y)=d(y,x) \quad \forall x,y \in X$
        \item $d(x,y)+d(y,z) \geq d(x,z) \quad \forall x,y,z \in X$
    \end{itemize}
\end{definition}
\begin{example}{\textbf Metric Topology}{}
    \noindent A set $X$ equipped with a metric $d$ is a metric space.\\
    Let $\beta$ be the set of all metric balls:
    \[B_d(x,r) = \{y \in X \mid d(x,y)<r\} \quad \t{for $x \in X$, $r>0$}\]
\end{example}
\begin{proposition}{}{}
    This is a basis for a topology
\end{proposition}
\begin{definition}{Metrizable}{}
    A topological space $X$ is called metrizable if $\exists$metric $d$ on $X$ inducing topology on $X$.
\end{definition}
\rmkb{There could be more than one metric inducing the same topology.}

\begin{example}{ \textbf Subspace Topology}{}
    Let $X$ be a set with topology $\T$. Let $Y \subseteq X$.\\
    Thus subspace topology on $Y$ is the collection of sets:
    \[\{U \cap Y \mid U \in \T\}\]
    
\end{example}
\begin{proposition}
    This is a topology on $Y$.
\end{proposition}
We call $Y$ a subspace of topology $X$.
\begin{example*}{}{}
    Suppose $(X,d)$ is a metric space. $Y \subseteq X$.\\
    The function $d$ is also a metric on $Y$. We have 2 toploogies on $Y$:
    \begin{itemize}
        \item Subspace topology as a subspace of X with metric topology.
        \item Metric topology on $Y$
    \end{itemize}
\end{example*}
\begin{proposition}{}{}
    These two topologies are the same.
\end{proposition}
\begin{proppf}
    Let $\T_\subseteq$ is subspace topology, and $\T_d$ is metric topology on $Y$.\\
    \[\T_d \t{ has a basis } \beta_d=\{B_d(y,r) \cap Y \mid y \in Y, \, r>0\}\]
    \[\T_\subseteq \t{ has a basis } \beta_\subseteq=\{B_d(x,r)\cap Y \mid x\in X, r>0\}\]
    Subspace Basis is just $X$'s basis but intersected with $Y$.\\
    Easy to see that $\beta_d \subseteq \beta_\subseteq$ thus $T_d \subseteq T_\subseteq$\\
    Let $y \in B \in \beta_\subseteq$, so $y \in B_d(x,r) \cap Y$ for some $x \in X$, $r>0$\\
    Using the triangle inequality, find some $r'>0$ s.t.
    \[B_d(y,r') \subseteq B_d(x,r)\]
    Then
    \[B_d(y,r)\cap Y \subseteq B_d(x,r)\cap Y\]
\end{proppf}

\begin{example*}{}
    Let $X,Y$ be nonempty topological spaces. Equip $X \times Y$ with the product topology.
    For each $y\in Y$, the set $X \times \{y\}$ is a subspace of $X \times Y$, and $X \times \{y\}$ is in bijection with $X$.\\
    Two topologies on $X$:
    \begin{itemize}
        \item Original topology
        \item Subspace topology on $X \times \{y\}$ identified with $X$.
    \end{itemize}
\end{example*}
\begin{proposition}{}{}
    These two topologies are the same.
\end{proposition}

\begin{example*}{}{}
    Let $X$ be a set with an order $<$. Let $Y \subseteq X$. Then $<$ gives an order relation of $Y$.\\
    Two topologies on $Y$:
    \begin{itemize}
        \item Order topology on $Y$
        \item Subspace topology
    \end{itemize}
\end{example*}
